            self._add_edge(SkillEdge(provider, "Docker", SkillRelation.PREREQUISITE))
    
    def get_implied_skills(self, skill: str) -> Set[str]:
        """
        Get all skills that are implied/covered by knowing the given skill.
        This is the CORE function that replaces hardcoded synonym prompts.
        
        Example: get_implied_skills("GitHub") → {"Git", "Version Control"}
        Example: get_implied_skills("React") → {"JavaScript", "Frontend", "TypeScript"}
        """
        implied = set()
        skill_lower = skill.lower()
        
        # Direct transfers
        for edge in self.edges:
            if edge.from_skill.lower() == skill_lower:
                if edge.relation == SkillRelation.TRANSFERS_TO:
                    implied.add(edge.to_skill)
                elif edge.relation == SkillRelation.SUPER_SKILL:
                    implied.add(edge.to_skill)
                    # Recursively get sub-skills of the super skill
                    implied.update(self.get_implied_skills(edge.to_skill))
        
        # Check aliases
        for name, node in self.nodes.items():
            if skill_lower in [a.lower() for a in node.aliases]:
                implied.update(self.get_implied_skills(name))
        
        return implied
    
    def is_covered_by(self, requested_skill: str, known_skills: List[str]) -> bool:
        """
        Check if a requested skill is covered by any known skill.
        
        Args:
            requested_skill: The skill being asked for
            known_skills: List of skills the user has
        
        Returns:
            True if any known skill covers the requested skill
        """
        requested_lower = requested_skill.lower()
        
        for known in known_skills:
            # Direct match
            if known.lower() == requested_lower:
                return True
            
            # Check if known skill implies the requested skill
            implied = self.get_implied_skills(known)
            for imp in implied:
                if imp.lower() == requested_lower:
                    return True
        
        return False
    
    def _normalize_skill(self, skill: str) -> Optional[str]:
        """Get canonical name for a skill, checking aliases"""
        skill_lower = skill.lower()
        
        for name, node in self.nodes.items():
            if name.lower() == skill_lower:
                return name
            if skill_lower in [a.lower() for a in node.aliases]:
                return name
        
        return None
    
    def add_skill(self, name: str, category: str, aliases: List[str] = None):
        """Programmatically add a new skill"""
        self._add_skill_if_missing(name, category)
        if aliases:
            self.nodes[name].aliases.update(aliases)

class ResumeAgent:
    """
    Agent 1: Extracts skills from resume with high precision.
    Key difference: Returns RAW skills only, normalization happens separately.
    """
    
    def __init__(self):
        self.llm = llm.with_structured_output(ResumeSkillsExtraction)
    
    def extract(self, state: AgentState) -> dict:
        """
        Extract raw skills from resume without any inference.
        This ensures we capture exactly what's mentioned.
        """
        system_prompt = """
        You are a precise technical skill extractor. Your job is to ONLY extract
        skills that are EXPLICITLY mentioned in the resume.
        
        RULES:
        1. Only list skills that are DIRECTLY mentioned (in skills section, 
           job descriptions, project descriptions, or technologies used)
        2. Do NOT infer or assume skills - only what is explicitly stated
        3. Include specific technologies: React, Node.js, PostgreSQL, AWS, etc.
        4. Include platforms: GitHub, npm, Docker Hub, etc.
        5. Include tools: Git, Docker, Kubernetes, etc.
        6. Include concepts only if explicitly mentioned: "machine learning",
           "blockchain", "microservices", etc.
        
        OUTPUT: A JSON with:
        - skills: List of exact skill names as they appear
        - projects: List of projects with their tech stacks
        - summary: Professional summary for job matching
        """
        
        try:
            result = self.llm.invoke([
                ("system", system_prompt),
                ("human", state["resume_text"])
            ])
            
            return {
                "raw_skills": result.skills,
                "profile": {
                    "projects": result.projects,
                    "summary": result.summary
                }
            }
        except Exception as e:
            return {"error": f"Resume extraction failed: {str(e)}"}
class SkillValidationAgent:
    """
    Agent 2: Validates and normalizes extracted skills using SKG.
    This is where the "GitHub → Git" inference happens programmatically.
    """
    
    def __init__(self, skill_graph: SkillKnowledgeGraph):
        self.skg = skill_graph
        self.normalizer = SkillNormalizer(skill_graph)
    
    def validate(self, state: AgentState) -> dict:
        """
        Process raw skills through SKG to:
        1. Normalize variations (nodejs → Node.js)
        2. Add implied skills (GitHub → Git)
        3. Mark confidence levels
        """
        raw_skills = state.get("raw_skills", [])
        covered = self.normalizer.get_covered_skills(raw_skills)
        
        # Build ExtractedSkill objects with confidence
        normalized_skills = []
        
        for skill in set(raw_skills):  # Dedupe
            canonical = self.normalizer.normalize(skill)
            implied = self.skg.get_implied_skills(canonical)
            
            # Check if this skill implies others
            if implied:
                confidence = SkillConfidence.IMPLIED
                source = "skill graph inference"
            else:
                confidence = SkillConfidence.EXPLICIT
                source = "resume"
            
            normalized_skills.append(ExtractedSkill(
                name=canonical,
                confidence=confidence,
                source=source
            ))
        
        # Add implied skills as DERIVED confidence
        for implied_skill in covered:
            if implied_skill not in [s.name for s in normalized_skills]:
                normalized_skills.append(ExtractedSkill(
                    name=implied_skill,
                    confidence=SkillConfidence.DERIVED,
                    source=f"implied from: {[s for s in raw_skills if self.normalizer.normalize(s) in self.skg.get_implied_skills(implied_skill)]}"
                ))
        
        return {
            "normalized_skills": normalized_skills,
            "covered_skills": covered
        }

            "competitive_edge": competitive_edge[:5]  # Top 5 competitive edge
        }
    
    def _get_covering_skills(self, target: str, covered: set) -> List[str]:
        """Find which covered skills cover the target skill"""
        covering = []
        for skill in covered:
            implied = self.skg.get_implied_skills(skill)
            if target.lower() in [i.lower() for i in implied]:
                covering.append(skill)
        return covering

class LearningResource(BaseModel):
    title: str
    url: str

    is_free: bool

    duration: Optional[str] = None
    source: str = "web_search"
class ResourceAgent:
    """
    Agent 4: Finds and validates learning resources via web search.
    This replaces hardcoded resource examples.
    """
    
    def __init__(self):
        self.search = TavilySearchResults(k=5)
        self.llm = ChatGroq(model="llama-3.3-70b-versatile", temperature=0.2)
    
    async def find_resources(self, skill: str, context: dict) -> List[LearningResource]:
        """
        Search the web for actual learning resources for a skill.
        Validates URLs before returning.
        """
        resources = []
        
        # Search queries optimized for finding actual resources
        queries = [
            f"best free resources to learn {skill} 2024",
            f"{skill} tutorial documentation official",
            f"best {skill} course for beginners"
        ]
        
        seen_urls = set()
        
        for query in queries:
            try:
                results = self.search.invoke({"query": query})
                
                for result in results:
                    url = result.get("url", "")
                    
                    # Dedupe by URL
                    if url in seen_urls:
                        continue
                    seen_urls.add(url)
                    
                    # Classify resource type
                    resource_type = self._classify_resource(url, result.get("content", ""))
                    
                    resources.append(LearningResource(
                        title=result.get("title", "Resource"),
                        url=url,
                        resource_type=resource_type,
                        is_free=self._is_free_resource(url, result.get("content", "")),
                        source="tavily_search"
                    ))
                
                # Limit to 4 resources per skill
                if len(resources) >= 4:
                    break
                    
            except Exception as e:
                print(f"Search error for {skill}: {e}")
                continue
        
        # Fallback to curated resources if search fails
        if not resources:
            resources = self._get_fallback_resources(skill)
        
        return resources[:4]  # Max 4 resources
    
    def _classify_resource(self, url: str, content: str) -> str:
        """Classify resource type from URL and content"""
        url_lower = url.lower()
        
        if "github.com" in url_lower or "gitlab.com" in url_lower:
            return "Practice"
        if any(x in url_lower for x in ["docs.", "documentation", ".io/docs"]):
            return "Documentation"
        if any(x in url_lower for x in ["youtube.com", "coursera", "udemy", "edx", "freecodecamp"]):
            return "Video"
        if any(x in url_lower for x in ["book", "leanpub", "amazon"]):
            return "Book"
        if "tutorial" in url_lower or "guide" in url_lower:
            return "Course"
        
        return "Course"  # Default
    
    def _is_free_resource(self, url: str, content: str) -> bool:
        """Determine if resource is free"""
        paid_keywords = ["buy", "price", "subscription", "premium", "paid"]
        
        # Known free platforms
        free_platforms = [
            "freecodecamp.org",
            "github.com",
            "docs.",
            "mozilla.org",
            "w3schools.com",
            "tutorialspoint.com",
            "youtube.com",
            "medium.com",
            "dev.to"
        ]
        
        if any(platform in url.lower() for platform in free_platforms):
            return True
        
        if any(keyword in content.lower() for keyword in paid_keywords):
            return False
        
        return True  # Assume free unless marked otherwise
    
    def _get_fallback_resources(self, skill: str) -> List[LearningResource]:

            ],
            "react": [
                LearningResource(
                    title="React Official Tutorial",
                    url="https://react.dev/learn",
                    resource_type="Documentation",
                    is_free=True
                )
            ],
            # Add fallbacks for common skills only
        }
        
        # Generic fallback for unknown skills
        return fallbacks.get(

class SkillGapAgent:

    """
    
    def __init__(self):
        self.skg = SkillKnowledgeGraph()
        self.resume_agent = ResumeAgent()

        self.market_agent = JobMarketAgent(self.skg)
        self.resource_agent = ResourceAgent()
        
        self._build_graph()
    
    def _build_graph(self):
        workflow = StateGraph(AgentState)
        
        # Register nodes
        workflow.add_node("extract_resume", self._extract_resume_node)
        workflow.add_node("validate_skills", self._validate_skills_node)
        workflow.add_node("match_jobs", self._match_jobs_node)
        workflow.add_node("analyze_gaps", self._analyze_gaps_node)
        workflow.add_node("find_resources", self._find_resources_node)
        workflow.add_node("generate_roadmap", self._generate_roadmap_node)
        workflow.add_node("validate_output", self._validate_output_node)
        
        # Entry point
        workflow.set_entry_point("extract_resume")
        
        # Normal flow
        workflow.add_edge("extract_resume", "validate_skills")
        workflow.add_edge("validate_skills", "match_jobs")
        workflow.add_edge("match_jobs", "analyze_gaps")
        workflow.add_edge("analyze_gaps", "find_resources")
        workflow.add_edge("find_resources", "generate_roadmap")
        workflow.add_edge("generate_roadmap", "validate_output")
        
        # Validation loop (self-correction)
        workflow.add_conditional_edges(
            "validate_output",
            self._should_retry,
            {
                "continue": END,
                "retry_extract": "extract_resume",
                "retry_validate": "validate_skills",
                "retry_gaps": "analyze_gaps"
            }
        )
        
        # Compile with memory for checkpointing
        self.app = workflow.compile(
            checkpointer=MemorySaver()
        )

                {"importance": gap["importance"], "context": state.get("profile", {})}
            )
        
        return {"resources_map": resources_map}

    def _validate_output_node(self, state: AgentState) -> dict:
        """
        Self-validation: Check if roadmap is accurate.
        This is where we catch issues like "Java in resume → Java roadmap".
        """
        corrections = []
        
        # Check 1: No roadmap for skills already in resume
        covered_skills = state.get("covered_skills", set())
        roadmap_skills = [step["skill"] for step in state.get("roadmap", {}).get("action_plan", [])]
        
        for skill in roadmap_skills:
            if self.skg.is_covered_by(skill, list(covered_skills)):
                corrections.append({
                    "type": "skill_already_known",
                    "skill": skill,
                    "covered_by": self._find_covering_skill(skill, covered_skills),
                    "action": "remove_from_roadmap"
                })
        
        # Check 2: Resources are valid URLs
        for skill, resources in state.get("resources_map", {}).items():
            for resource in resources:
                if not self._validate_url(resource["url"]):
                    corrections.append({
                        "type": "invalid_url",
                        "skill": skill,
                        "url": resource["url"],
                        "action": "replace_resource"
                    })
        
        return {
            "corrections": corrections,
            "validation_passed": len(corrections) == 0
        }
    
    def _should_retry(self, state: AgentState) -> Literal["continue", "retry_extract", "retry_validate", "retry_gaps"]:
        """Determine if retry is needed based on corrections"""
        if state.get("validation_passed", False):
            return "continue"
        
        corrections = state.get("corrections", [])
        
        # Categorize corrections by type
        types = [c["type"] for c in corrections]
        
        if "skill_already_known" in types:
            # Remove the incorrectly added skills and continue
            return "retry_gaps"
        
        if "invalid_url" in types:
            # Try to find replacement resources
            return "continue"  # Or implement specific retry
        
        return "continue"
    
    # Helper methods
    def _find_covering_skill(self, target: str, covered: set) -> str:
        for skill in covered:
            implied = self.skg.get_implied_skills(skill)
            if target.lower() in [i.lower() for i in implied]:
                return skill
        return target
    
    def _validate_url(self, url: str) -> bool:

from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np
class FuzzySkillMatcher:
    """
    Uses embeddings to match skill names with fuzzy matching.
    Handles cases like "NodeJS" vs "Node.js" that normalization might miss.
    """
    
    def __init__(self, skill_graph: SkillKnowledgeGraph):
        self.skg = skill_graph
        self._build_skill_embeddings()
    
    def _build_skill_embeddings(self):
        """Pre-compute embeddings for all skill names and aliases"""
        all_skills = []
        
        for name, node in self.skg.nodes.items():
            all_skills.append(name)
            all_skills.extend(node.aliases)
        
        # Using simple TF-IDF for now (can upgrade to actual embeddings)
        self.vectorizer = TfidfVectorizer(analyzer='char_wb', ngram_range=(2, 4))
        self.skill_vectors = self.vectorizer.fit_transform(all_skills)
        self.all_skills = all_skills
    
    def find_similar(self, query: str, threshold: float = 0.7) -> List[tuple]:
        """Find skills similar to query"""
        query_vec = self.vectorizer.transform([query])
        similarities = cosine_similarity(query_vec, self.skill_vectors)[0]
        
        results = []
        for idx, score in enumerate(similarities):
            if score >= threshold:
                results.append((self.all_skills[idx], score))
        
        return sorted(results, key=lambda x: x[1], reverse=True)
